> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add EAP — gradient-based approximate edge attribution over a transformer's residual-stream computation graph — as v1.0 patching sub-spec 2, built on the shipped core intervention primitive.

**Architecture:** A model-agnostic scoring *engine* (graph enumeration + stacked-cache einsum scoring) separated from model-specific *caching* (writer activations + reader-slot gradients). Correctness is proven incrementally: the engine on a linear MLP-only toy (exact cross-check vs brute-force `patch_site`), then attention edges, then the realistic HF backend, then TL, then EAP-IG.

**Tech Stack:** Python 3.12, PyTorch, HF transformers (eager, Llama-family), TransformerLens (optional, lazy).

# src/circuitry/patching/graph.py
"""EAP edge graph: nodes, edges, causal enumeration. Design spec §2."""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Literal

import torch
from torch import Tensor

Slot = Literal["q", "k", "v", "mlp_in", "logits_in"]
_ATTN_SLOTS: tuple[Slot, ...] = ("q", "k", "v")


@dataclass(frozen=True)
class Node:
    kind: str                 # "embed" | "attn_head" | "mlp" | "logits"
    layer: int | None = None
    head: int | None = None


@dataclass(frozen=True)
class Edge:
    writer: Node
    reader: Node
    slot: Slot


def _order(node: Node, n_layers: int) -> int:
    """Forward-order rank. Heads in one attn block share a rank (parallel)."""
    if node.kind == "embed":
        return 0
    if node.kind == "attn_head":
        return 1 + 2 * node.layer
    if node.kind == "mlp":
        return 2 + 2 * node.layer
    if node.kind == "logits":
        return 1 + 2 * n_layers
    raise ValueError(f"unknown node kind {node.kind!r}")


@dataclass
class EdgeGraph:
    n_layers: int
    n_heads: int
    writers: list[Node]
    readers: list[tuple[Node, Slot]]
    edges: list[Edge]
    _w_idx: dict[Node, int] = field(default_factory=dict)
    _r_idx: dict[tuple[Node, Slot], int] = field(default_factory=dict)

    def __post_init__(self) -> None:
        self._w_idx = {n: i for i, n in enumerate(self.writers)}
        self._r_idx = {rs: i for i, rs in enumerate(self.readers)}

    def writer_index(self, node: Node) -> int:
        return self._w_idx[node]

    def reader_index(self, node: Node, slot: Slot) -> int:
        return self._r_idx[(node, slot)]

    def valid_mask(self) -> Tensor:
        mask = torch.zeros(len(self.writers), len(self.readers), dtype=torch.bool)
        for e in self.edges:
            mask[self.writer_index(e.writer), self.reader_index(e.reader, e.slot)] = True
        return mask


def build_graph(n_layers: int, n_heads: int) -> EdgeGraph:
    writers: list[Node] = [Node("embed")]
    readers: list[tuple[Node, Slot]] = []
    for L in range(n_layers):
        for h in range(n_heads):
            head = Node("attn_head", L, h)
            writers.append(head)
            for slot in _ATTN_SLOTS:
                readers.append((head, slot))
        mlp = Node("mlp", L)
        writers.append(mlp)
        readers.append((mlp, "mlp_in"))
    logits = Node("logits")
    readers.append((logits, "logits_in"))

    edges: list[Edge] = []
    for w in writers:
        wo = _order(w, n_layers)
        for rnode, slot in readers:
            if wo < _order(rnode, n_layers):
                edges.append(Edge(w, rnode, slot))
    return EdgeGraph(n_layers, n_heads, writers, readers, edges)
```

# src/circuitry/patching/eap.py
"""EAP scoring engine + runner. Design spec §3, §5, §7."""
from __future__ import annotations

from dataclasses import dataclass

import torch
from torch import Tensor

from circuitry.patching.graph import Edge, EdgeGraph


@dataclass
class EAPResult:
    graph: EdgeGraph
    scores: dict[Edge, float]

    def ranked(self) -> list[tuple[Edge, float]]:
        return sorted(self.scores.items(), key=lambda kv: abs(kv[1]), reverse=True)

    def top_k(self, n: int) -> list[tuple[Edge, float]]:
        return self.ranked()[:n]

    def threshold(self, tau: float) -> list[Edge]:
        return [e for e, s in self.scores.items() if abs(s) >= tau]


def score_edges(
    graph: EdgeGraph,
    act_clean: Tensor,        # (batch, pos, |writers|, d_model)
    act_corrupted: Tensor,    # (batch, pos, |writers|, d_model)
    grad_clean: Tensor,       # (batch, pos, |readers|, d_model)
) -> EAPResult:
    """Analytic EAP scoring: score[w,r] = sum_{b,p,d} (act_corr-act_clean)[w] * grad[r]."""
    delta = act_corrupted - act_clean
    # (|writers|, |readers|): sum over batch, pos, d_model
    score_mat = torch.einsum("bpwd,bprd->wr", delta, grad_clean)
    scores: dict[Edge, float] = {}
    for e in graph.edges:
        w = graph.writer_index(e.writer)
        r = graph.reader_index(e.reader, e.slot)
        scores[e] = float(score_mat[w, r].item())
    return EAPResult(graph, scores)
```

**`LinearMLPToy` design (add to conftest.py):** residual stream of width `d`; `n_layers` linear "MLP" blocks each computing `down(act(up(x)))` but with `act = identity` (pure linear) and identity-ish LayerNorm (omit LN entirely). `embed` is a linear map from one-hot token ids → `d`. Output = residual @ unembed. Module names mirror HF (`model.layers.{L}.mlp.up_proj` / `down_proj`, `model.embed_tokens`, `lm_head`) so the same addressing works later. Forward returns logits `(batch, seq, vocab)`.

```python
# add to tests/patching/conftest.py
class LinearMLPToy(nn.Module):
    """Attention-free linear residual stack with HF-like names. Fully linear:
    EAP's first-order score is EXACT here, so it must equal brute-force patching."""

    def __init__(self, n_layers=2, d=4, vocab=5):
        super().__init__()
        self.n_layers, self.d, self.vocab = n_layers, d, vocab
        self.embed_tokens = nn.Embedding(vocab, d)
        self.layers = nn.ModuleList()
        for _ in range(n_layers):
            block = nn.Module()
            block.mlp = nn.Module()
            block.mlp.up_proj = nn.Linear(d, d, bias=False)
            block.mlp.down_proj = nn.Linear(d, d, bias=False)
            self.layers.append(block)
        self.lm_head = nn.Linear(d, vocab, bias=False)
        for m in self.modules():
            if isinstance(m, nn.Linear):
                nn.init.normal_(m.weight, std=0.5)

    def forward(self, input_ids: torch.Tensor) -> torch.Tensor:
        x = self.embed_tokens(input_ids)            # (b, s, d) — embed writer output
        for block in self.layers:
            mlp_out = block.mlp.down_proj(block.mlp.up_proj(x))   # linear, no activation
            x = x + mlp_out                          # residual write
        return self.lm_head(x)                       # (b, s, vocab)


@pytest.fixture
def linear_mlp_toy():
    torch.manual_seed(0)
    return LinearMLPToy(n_layers=2, d=4, vocab=5)

# tests/patching/test_eap_mlp_exact.py
"""EAP exact cross-check on a linear MLP-only toy: analytic scores == brute-force
patch_site edge patching. Spec §8 (the correctness gate)."""
from __future__ import annotations

import pytest
import torch

from circuitry.core.patching import logit_diff
from circuitry.patching.eap import EAPRunner


def _metric(logits):
    return logit_diff(logits, correct=0, incorrect=1)


def test_eap_node_aggregate_matches_node_patching(linear_mlp_toy):
    """On a fully linear model, brute-force NODE patching of writer u changes the
    metric by EXACTLY the sum of EAP edge scores leaving u — because the gradient
    is path-constant and u broadcasts one residual contribution to all its
    readers. This is the exact, tractable correctness gate (node patching uses
    the shipped patch_site; true per-edge patching is not required)."""
    from collections import defaultdict

    torch.manual_seed(0)
    clean = torch.tensor([[0, 1, 2]])
    corrupted = torch.tensor([[2, 0, 1]])

    runner = EAPRunner(linear_mlp_toy)   # MLP-only: backend inferred, no resolver needed
    result = runner.run(clean_inputs=clean, corrupted_inputs=corrupted, metric=_metric)

    # brute force: patch each writer node's contribution clean->corrupted, measure Δmetric
    node_delta = runner.bruteforce_node_scores(
        clean_inputs=clean, corrupted_inputs=corrupted, metric=_metric
    )
    agg: dict = defaultdict(float)
    for edge, s in result.scores.items():
        agg[edge.writer] += s
    assert set(agg) == set(node_delta)
    for node, delta in node_delta.items():
        assert agg[node] == pytest.approx(delta, abs=1e-4), node


def test_model_clean_after_eap(linear_mlp_toy):
    clean = torch.tensor([[0, 1, 2]])
    before = linear_mlp_toy(clean).clone()
    runner = EAPRunner(linear_mlp_toy)
    runner.run(clean_inputs=clean, corrupted_inputs=torch.tensor([[2, 0, 1]]), metric=_metric)
    after = linear_mlp_toy(clean)
    assert torch.equal(before, after)
```

- [ ] **Step 3: Implement `EAPRunner` (MLP + embed nodes only)**

Add to `src/circuitry/patching/eap.py`. The runner: (1) builds the graph from the model's `n_layers`/`n_heads`; (2) caches per-node writer activations (`d_model`) and per-reader-slot gradients on clean & corrupted runs via forward/backward hooks; (3) stacks into `(b,p,|W|,d)` / `(b,p,|R|,d)`; (4) calls `score_edges`. For this task, only `embed`, `mlp` writers and `mlp_in`, `logits_in` readers exist (n_heads=0 path), so no attention handling yet.

Key mechanics:
- **embed writer** = output of `model.embed_tokens` (forward hook, capture output).
- **mlp writer** = output of `model.layers.{L}.mlp` (the residual *contribution*; in the toy that's `down_proj` output — hook the `mlp` submodule output).
- **mlp_in reader** = gradient w.r.t. the residual input to `model.layers.{L}.mlp`. Capture via a forward hook that calls `.requires_grad_(True).retain_grad()` on the input (or registers a tensor hook), then read `.grad` after backward.
- **logits_in reader** = gradient w.r.t. the residual entering `lm_head` (hook `lm_head` input).
- Run clean forward, `metric(logits).backward()`, collect gradients + clean acts. Run corrupted forward (no grad needed) for corrupted acts. `delta = corrupted - clean`.
- **frozen-model discipline**: reuse the same pattern as `patch_site` (params `requires_grad=False` saved/restored inside try/finally; eval mode managed; hooks removed on exit).
- `bruteforce_node_scores(...) -> dict[Node, float]`: for each writer node, patch that node's residual *contribution* from clean to corrupted and measure `metric(patched) − metric(clean)`. Patch via `patch_site` on the node's corresponding `Site` (mlp → `mlp_out`; embed → special-case forward hook on `embed_tokens`, since `Site` has no embed component), using the node's corrupted activation captured during the corrupted run. This is node-level patching only — exactly what `patch_site` provides.

> **Why node-aggregate is exact:** on a linear model the metric is linear in each node's residual contribution, and a writer broadcasts the *same* contribution to all readers; so patching node `u` clean→corrupted changes the metric by `Σ_readers grad_r · Δact_u = Σ_r EAP(u→r)`. The test asserts exactly this. **Implementer note:** capturing residual-input gradients is easy to get subtly wrong — the test is the spec; iterate until it passes to `abs=1e-4`. If wiring is fiddly, first get `n_layers=1` passing (writers: embed, mlp0; readers: mlp_in@0, logits_in), then generalize.

Provide `EAPRunner.__init__(self, model, resolver=None)` (resolver unused for the MLP path; required later). Implement `run(...)` returning `EAPResult` and `bruteforce_node_scores(...)` returning `dict[Node, float]` (per-writer-node metric delta from node patching).

**`LinearAttnToy`:** like `LinearMLPToy` but each layer has a `self_attn` with `q_proj`/`k_proj`/`v_proj`/`o_proj` and a **fixed (non-learned, input-independent) attention pattern** so the head output is linear in the residual. Names mirror HF. Because the pattern is fixed, the whole model is linear → the node-aggregate gate is exact.

> **Linearity caveat:** with a fixed pattern, `q`/`k` do **not** affect the output, so their *reader* gradients are ~0 and `q`/`k` incoming-edge scores are ~0 (in both EAP and brute force — they agree trivially). The exact gate here validates writer `z@W_O`, the `v` reader path, MLP edges, and scoring. The `q`/`k`/`v` reader back-map *itself* is validated independently by a targeted hand-computed test in Task 5 (`test_qkv_backmap_matches_reference`) and by the HF rank correlation.

# add to tests/patching/conftest.py
class LinearAttnToy(nn.Module):
    """Linear transformer with a FIXED, input-independent attention pattern, so
    the whole model is linear and the EAP node-aggregate gate is exact. HF-like
    module names. q_proj/k_proj exist (for resolver hooks) but don't affect the
    output (pattern is fixed) — see the linearity caveat in the plan."""

    def __init__(self, n_layers=2, n_heads=2, d=4, vocab=5, seq=4):
        super().__init__()
        self.n_layers, self.n_heads, self.d, self.vocab = n_layers, n_heads, d, vocab
        self.head_dim = d // n_heads
        self.embed_tokens = nn.Embedding(vocab, d)
        self.layers = nn.ModuleList()
        for _ in range(n_layers):
            layer = nn.Module()
            attn = nn.Module()
            attn.q_proj = nn.Linear(d, d, bias=False)
            attn.k_proj = nn.Linear(d, d, bias=False)
            attn.v_proj = nn.Linear(d, d, bias=False)
            attn.o_proj = nn.Linear(d, d, bias=False)
            layer.self_attn = attn
            mlp = nn.Module()
            mlp.up_proj = nn.Linear(d, d, bias=False)
            mlp.down_proj = nn.Linear(d, d, bias=False)
            layer.mlp = mlp
            self.layers.append(layer)
        self.lm_head = nn.Linear(d, vocab, bias=False)
        pat = torch.tril(torch.ones(seq, seq))
        self.register_buffer("attn_pattern", pat / pat.sum(-1, keepdim=True))  # fixed
        torch.manual_seed(0)
        for m in self.modules():
            if isinstance(m, nn.Linear):
                nn.init.normal_(m.weight, std=0.4)

    def forward(self, input_ids):
        b, s = input_ids.shape
        x = self.embed_tokens(input_ids)
        for layer in self.layers:
            v = layer.self_attn.v_proj(x).view(b, s, self.n_heads, self.head_dim)
            attended = torch.einsum("st,bthd->bshd", self.attn_pattern[:s, :s], v)
            attn_out = layer.self_attn.o_proj(attended.reshape(b, s, self.d))
            x = x + attn_out
            x = x + layer.mlp.down_proj(layer.mlp.up_proj(x))
        return self.lm_head(x)

- **Back-map helper** (`_backmap_qkv_grad`, the orientation pinned by `test_qkv_backmap_matches_reference`):

```python
def _backmap_qkv_grad(dL_dq_head: Tensor, W_proj_head: Tensor, ln_scale: Tensor) -> Tensor:
    """Map a per-head projection-output gradient to residual (d_model) space.
    dL_dq_head: (b, p, head_dim); W_proj_head: (head_dim, d_model) slice of the
    q/k/v weight; ln_scale: (b, p, 1) stop-gradient RMSNorm scale.
    Returns (b, p, d_model)."""
    return (dL_dq_head @ W_proj_head) * ln_scale
```

  Note `W_proj_head` is the `(head_dim, d_model)` slice of the projection weight — PyTorch stores `Linear.weight` as `(out, in) = (n_heads*head_dim, d_model)`, so head `h`'s slice is `weight[h*head_dim:(h+1)*head_dim, :]`, already `(head_dim, d_model)`; no transpose.
- **LN scale (two-hook):** forward hook on each layer's pre-attention norm (`model.layers.{L}.input_layernorm`) capturing its *input* `x_resid`; compute the RMSNorm scale `1/sqrt(mean(x_resid^2) + eps)` per (batch,pos), kept as a stop-gradient constant, passed as `ln_scale`. (Llama uses RMSNorm; the pre-MLP norm is `post_attention_layernorm` for `mlp_in`.)
- **GQA:** `_kv_head_for(query_head, n_heads, n_kv_heads) = query_head // (n_heads // n_kv_heads)`. For `k`/`v` reader slots, back-map through the kv-head's projection slice.
- **Clean error:** if the resolver can't find the expected modules (non-Llama), raise a clear `ValueError`.

> **Implementer note:** the `abs`-exact gate from Task 4 no longer applies (real softmax/RMSNorm). The HF correctness anchors are: (a) the rank-correlation test, (b) the LN-scale and GQA unit tests, (c) the Task-4 exact test still passing (the engine math is unchanged). If correlation is < 0.7, the back-map or LN handling is wrong — debug, don't loosen.
