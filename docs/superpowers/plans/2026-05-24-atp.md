# AtP\* (Attribution Patching with corrections) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add AtP\* node attribution (vanilla AtP + QK fix + GradDrop + neuron-level nodes + a `verify_top_k` calibration helper) as v1.0 patching sub-spec 3.

**Architecture:** A new `AtPRunner` reusing EAP's writer-activation caching, backends (HF eager / TL), `_t` metrics, GQA + module-location logic. Vanilla AtP scores nodes via `Δact · full-grad` (gradient w.r.t. the node's OWN output, via `retain_grad` — not EAP's component-only clones). The QK fix recomputes attention for q/k nodes. Correctness is proven incrementally: vanilla AtP exact on a linear toy (vs brute-force `patch_site` per node), then the QK fix gated by real-model brute-force comparison, then GradDrop, then verification.

**Tech Stack:** Python 3.12, PyTorch, HF transformers (eager, Llama-family; `output_attentions` for the QK pattern), TransformerLens (optional, lazy).

```python
# src/circuitry/patching/atp.py
"""AtP* node attribution. Design spec docs/superpowers/specs/2026-05-24-atp-design.md."""
from __future__ import annotations

from dataclasses import dataclass

from circuitry.patching.graph import Node

_ATTN_SLOTS = ("q", "k", "v")


@dataclass(frozen=True)
class AtPNode:
    """An attributable activation: a graph Node plus an optional attention slot."""
    node: Node
    slot: str | None = None  # "q"/"k"/"v" for attn_head, else None


def enumerate_nodes(n_layers: int, n_heads: int, d_mlp: int | None = None) -> list[AtPNode]:
    nodes: list[AtPNode] = [AtPNode(Node("embed"), None)]
    for L in range(n_layers):
        for h in range(n_heads):
            for slot in _ATTN_SLOTS:
                nodes.append(AtPNode(Node("attn_head", L, h), slot))
        nodes.append(AtPNode(Node("mlp", L), None))
        if d_mlp is not None:
            for nidx in range(d_mlp):
                nodes.append(AtPNode(Node("mlp_neuron", L, neuron=nidx), None))
    return nodes


@dataclass
class AtPResult:
    scores: dict[AtPNode, float]

    def ranked(self) -> list[tuple[AtPNode, float]]:
        return sorted(self.scores.items(), key=lambda kv: abs(kv[1]), reverse=True)

    def top_k(self, n: int) -> list[tuple[AtPNode, float]]:
        return self.ranked()[:n]

    def threshold(self, tau: float) -> list[AtPNode]:
        return [n for n, s in self.scores.items() if abs(s) >= tau]
```

The field is unused by `build_graph` (EAP has no neuron nodes), so EAP tests stay green — verify with `venv/bin/pytest tests/patching/test_graph.py -q` after the edit.

**Context:** Vanilla AtP: `score(node) = Σ(Δact · grad)` over the node's activation dims. `Δact = corrupted_act − clean_act`; `grad = ∂metric/∂act` (FULL gradient via `retain_grad` on the node's activation tensor — NOT a component-only clone). On a linear model this equals the brute-force `patch_site` metric delta for that node. This task does v/mlp/embed nodes (NO neurons, NO q/k — those are Tasks 3 and 5). Reuse EAP's `_locate_layers`/`_embed`, `_call_model`, frozen-model discipline.

Activation/grad capture points (per node):
- `embed` → `embed_tokens` output (d_model).
- `mlp(L)` → `layers[L].mlp` output (d_model).
- `attn_head(L,h)` slot `v` → `layers[L].self_attn.v_proj` output, head-h slice (d_head).

```python
# tests/patching/test_atp_vanilla_exact.py
"""Vanilla AtP exact gate: Δact·grad == brute-force patch_site per node, on a
linear model. Independent ground truth (real forward intervention)."""
from __future__ import annotations

import pytest
import torch

from circuitry.core.patching import logit_diff_t
from circuitry.patching.atp import AtPNode, AtPRunner
from circuitry.patching.graph import Node
from circuitry.patching.sites import HFSiteResolver


def _metric(logits):
    return logit_diff_t(logits, correct=0, incorrect=1)


def _resolver(toy):
    return HFSiteResolver(n_heads=toy.n_heads, d_model=toy.d, d_mlp=toy.d,
                          layer_pattern="layers.{L}")


def test_vanilla_atp_matches_bruteforce_per_node(linear_attn_toy):
    """v / mlp / embed node scores == brute-force patch_site metric delta (exact
    on a linear model). q/k excluded here (Task 5)."""
    torch.manual_seed(0)
    clean = torch.tensor([[0, 1, 2, 3]])
    corrupted = torch.tensor([[3, 2, 1, 0]])
    runner = AtPRunner(linear_attn_toy, _resolver(linear_attn_toy))
    result = runner.run(clean_inputs=clean, corrupted_inputs=corrupted,
                        metric=_metric, neurons=False)
    bf = runner.bruteforce_node_scores(clean_inputs=clean, corrupted_inputs=corrupted,
                                       metric=_metric, nodes=list(result.scores))
    # check v / mlp / embed nodes exactly; q/k are ~0 on the fixed-pattern toy
    for node, score in result.scores.items():
        if node.slot in ("q", "k"):
            continue
        assert score == pytest.approx(bf[node], abs=1e-4), node


def test_model_clean_after_atp(linear_attn_toy):
    clean = torch.tensor([[0, 1, 2, 3]])
    before = linear_attn_toy(clean).clone()
    runner = AtPRunner(linear_attn_toy, _resolver(linear_attn_toy))
    runner.run(clean_inputs=clean, corrupted_inputs=torch.tensor([[3, 2, 1, 0]]),
               metric=_metric, neurons=False)
    assert torch.equal(linear_attn_toy(clean), before)
```

- [ ] **Step 3: Implement `AtPRunner` (vanilla, v/mlp/embed)**

Add `AtPRunner` to `atp.py`. Structure mirrors `EAPRunner` (read `eap.py`):
- `__init__(self, model, resolver)`: derive `n_layers`/`n_heads` (from resolver, or `model.cfg` for TL); store. Use EAP's locator approach (import or replicate `_locate_layers`/`_embed`; if cleanly factorable, move those to a shared helper module imported by both — but DO NOT regress EAP).
- `run(clean_inputs, corrupted_inputs, metric, *, neurons=False, graddrop=False)`:
  1. Freeze params / eval (save+restore in finally), like `patch_site`.
  2. Corrupted forward (no grad): cache node activations (embed/mlp outputs, per-head v).
  3. Clean forward: `retain_grad` on the SAME node-activation tensors (full gradient — register the activation tensor, NOT a clone), `metric(logits).backward()`, read `.grad`.
  4. `score(node) = float((Δact * grad).sum().item())` per node (sum over all the activation's non-batch... actually sum over EVERYTHING: positions and feature dims — AtP attributes the node's total patch effect summed over positions by default).
  5. Return `AtPResult({node: score})` for v/mlp/embed nodes (q/k filled by Task 5; for now compute them too via the QK path placeholder OR leave q/k scored as vanilla Δact·grad in head-output space — but the test skips q/k, so a placeholder of 0.0 for q/k is acceptable here; Task 5 replaces it).
- `bruteforce_node_scores(clean_inputs, corrupted_inputs, metric, nodes)`: for each node, patch its activation clean→corrupted via `patch_site` (or a direct forward hook replacing the node activation with its cached corrupted value), run clean inputs, measure `metric(patched).item() − metric(clean).item()`. For v use the per-head v_proj-output slice; mlp the mlp output; embed the embed output. Return `{node: delta}`. This is independent ground truth — a real forward intervention.

> **Implementer note:** vanilla AtP grad is the FULL gradient (retain_grad on the node output, no clone). The exact gate is the spec; iterate until v/mlp/embed match brute-force at 1e-4. If they don't, the grad capture point ≠ the brute-force patch point — align them. Do NOT loosen the tolerance.

### Task 5: QK fix scoring (HIGHEST RISK)

**Files:** Modify `atp.py`; Create `tests/patching/test_atp_qk.py`.

**Context (spec §4):** for head h, query node (key symmetric):
1. `Δq_h = q_corrupted_h − q_clean_h` (per-head, d_head).
2. `pattern' = softmax((Q with head h's query replaced by q_corrupted_h) · K_clean^T / √d_head + causal_mask)`; `Δpattern_h = pattern'_h − pattern_clean_h`.
3. `Δz_h = Δpattern_h @ V_clean_h` (d_head).
4. `Δhead_out_h = Δz_h @ W_O[h]` (d_model; `W_O[h] = o_proj.weight[:, h*hd:(h+1)*hd]`).
5. `score = Σ_d_model (Δhead_out_h ⊙ grad_head_out_h)`, where `grad_head_out_h = ∂metric/∂(head h's d_model residual contribution = z@W_O)` — the SAME activation/gradient EAP captures as the attention writer. Capture it via `retain_grad` on the per-head `z@W_O` contribution (reuse EAP's writer-side gradient machinery).

The clean attention pattern + V + Q + K come from the clean forward: on HF use `output_attentions=True` (eager; reuse v0.9.2 `config.output_attentions`) for `pattern_clean`, and `q_proj`/`k_proj`/`v_proj` outputs (per head, GQA-mapped) for Q/K/V; on TL use `hook_pattern`/`hook_q`/`hook_k`/`hook_v`. q_corrupted_h comes from the corrupted forward's `q_proj` output.

**The gate (linear toy can't test the QK fix — softmax is what it corrects):** on a tiny REAL HF Llama, the QK-fixed q/k score must approximate brute-force `patch_site(attn_head_q_out / k_out)` BETTER than vanilla `Δq·grad_q` does.

```python
# tests/patching/test_atp_qk.py
"""QK fix gated on a real HF Llama: QK-fixed q/k attribution approximates
brute-force patch_site(q/k) better than vanilla Δq·grad. Anti-stub."""
from __future__ import annotations

import numpy as np
import pytest
import torch

transformers = pytest.importorskip("transformers")
from circuitry.core.patching import logit_diff_t
from circuitry.patching.atp import AtPRunner
from circuitry.patching.sites import HFSiteResolver


def _tiny_llama():
    cfg = transformers.LlamaConfig(
        vocab_size=32, hidden_size=16, intermediate_size=32, num_hidden_layers=2,
        num_attention_heads=4, num_key_value_heads=4, max_position_embeddings=16)
    torch.manual_seed(0)
    m = transformers.LlamaForCausalLM(cfg).eval()
    m.config._attn_implementation = "eager"
    return m


def _m(out):
    logits = out.logits if hasattr(out, "logits") else out
    return logit_diff_t(logits, correct=0, incorrect=1)


def test_qk_fix_beats_vanilla_against_bruteforce():
    model = _tiny_llama()
    resolver = HFSiteResolver(n_heads=4, d_model=16, d_mlp=32, layer_pattern="model.layers.{L}")
    runner = AtPRunner(model, resolver)
    clean = {"input_ids": torch.tensor([[1, 2, 3, 4]])}
    corrupted = {"input_ids": torch.tensor([[4, 3, 2, 1]])}

    qk_fixed = runner.run(clean_inputs=clean, corrupted_inputs=corrupted, metric=_m)
    vanilla = runner.run(clean_inputs=clean, corrupted_inputs=corrupted, metric=_m,
                         qk_fix=False)
    bf = runner.bruteforce_node_scores(clean_inputs=clean, corrupted_inputs=corrupted,
                                       metric=_m, nodes=list(qk_fixed.scores))

    qk_nodes = [n for n in qk_fixed.scores if n.slot in ("q", "k")]
    truth = np.array([bf[n] for n in qk_nodes])
    fixed = np.array([qk_fixed.scores[n] for n in qk_nodes])
    plain = np.array([vanilla.scores[n] for n in qk_nodes])
    # QK-fixed is closer to ground truth than vanilla (lower error)
    assert np.abs(fixed - truth).sum() < np.abs(plain - truth).sum()
    # and reasonably correlated with truth
    if np.std(fixed) > 1e-9 and np.std(truth) > 1e-9:
        assert np.corrcoef(fixed, truth)[0, 1] > 0.5


def test_qk_operand_space_match():
    """Δhead_out and grad_head_out are both d_model before the dot product."""
    model = _tiny_llama()
    resolver = HFSiteResolver(n_heads=4, d_model=16, d_mlp=32, layer_pattern="model.layers.{L}")
    runner = AtPRunner(model, resolver)
    clean = {"input_ids": torch.tensor([[1, 2, 3, 4]])}
    corrupted = {"input_ids": torch.tensor([[4, 3, 2, 1]])}
```

```python
"""GradDrop reduces sign-cancellation undercounting."""
from __future__ import annotations

import pytest
import torch

from circuitry.core.patching import logit_diff_t
from circuitry.patching.atp import AtPRunner
from circuitry.patching.sites import HFSiteResolver


def _metric(logits):
    return logit_diff_t(logits, correct=0, incorrect=1)


def test_graddrop_geq_plain_magnitude(linear_mlp_toy):
    # GradDrop sums |per-position score|, so its magnitude is >= |sum of per-position
    # scores| (plain AtP) for every node — strictly greater when signs flip across pos.
    clean = torch.tensor([[0, 1, 2]])
    corrupted = torch.tensor([[2, 0, 1]])
    resolver = HFSiteResolver(n_heads=1, d_model=linear_mlp_toy.d, d_mlp=linear_mlp_toy.d,
                              layer_pattern="layers.{L}")
    runner = AtPRunner(linear_mlp_toy, resolver)
    plain = runner.run(clean_inputs=clean, corrupted_inputs=corrupted, metric=_metric, neurons=False)
    gd = runner.run(clean_inputs=clean, corrupted_inputs=corrupted, metric=_metric,
                    neurons=False, graddrop=True)
    for node in plain.scores:
        if node.slot in ("q", "k"):
            continue
        assert abs(gd.scores[node]) >= abs(plain.scores[node]) - 1e-5, node
```
